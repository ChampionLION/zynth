The 5 Questions
Q1 — Maps: Domain + Execution pattern
"You have $10,000 land in your account with a note: 'spend it in 48 hours or lose it.' What actually happens?"
Options:
A — I buy tools, equipment, or resources for something I've been wanting to build
B — I research obsessively for 47 hours and spend it in the last hour on exactly the right thing
C — I invest it into something that will grow — money should make more money
D — I create an experience — travel, document everything, make it unforgettable
E — I split it between people I care about and something that matters to me
F — I freeze. 48 hours passes. I probably lose it.
Signal map: A=Technical/builder+high execution, B=Analytical+overthinking weakness, C=Strategic+long-term, D=Creative+experiential, E=Social domain, F=Low execution+fear_of_failure

Q2 — Maps: Learning style + Behavior signature
"A fire alarm goes off in a building you're in. False alarm. What were you doing before it pulled you away?"
Options:
A — Making or building something — I was deep in it and annoyed at the interruption
B — Reading, watching, or consuming something that caught my attention earlier
C — Talking to someone — it was actually a good conversation
D — Organizing something, planning, mapping out an idea in my head or on paper
E — On my phone, not doing much — I needed a break anyway
F — Working toward something I set for myself — I was in a rare focused state
Signal map: A=Practical learner+Technical/Creative, B=Analytical learner+curiosity, C=Social+Guided learner, D=Guided+Strategic, E=Inconsistency weakness, F=Exploratory+high consistency

Q3 — Maps: Hidden strengths + Weakness signal + Domain
"If you could wake up tomorrow with one of these and use it every single day — which one do you actually choose?"
Options:
A — The ability to learn any skill perfectly in one focused hour
B — Knowing exactly why anything works the way it does, instantly
C — Whatever I imagine comes out perfectly in real life — exactly as I pictured it
D — Anyone I speak to understands me completely, no miscommunication ever
E — I can see the single best decision to make in any situation before I make it
F — I never forget anything I've ever experienced or learned
Signal map: A=Execution-driven+Practical, B=Analytical+insight, C=Creative+imagination, D=Social+communication, E=Strategic+decision, F=Analytical+memory_crutch weakness

Q4 — Maps: Consistency + Execution + Weakness signal

Signal map: A=High consistency+execution+flow, B=Exploratory+scattered_interests weakness, C=Guided+moderate consistency, D=Poor_focus+low execution, E=Social+externally motivated, F=Inconsistency weakness+starter-not-finisher

Quest Type Resolver
export function resolveQuestType(profile) {
  const { quest_history, streak, level } = profile
  // First quest ever → calibration
  if (!quest_history || quest_history.length === 0) return 'calibration'
  // Three consecutive Fails → weakness quest
  const lastThree = quest_history.slice(-3)
  if (lastThree.length === 3 && lastThree.every(q => q.grade === 'Fail')) {
    return 'weakness'
  }
  // Weekly mission — streak multiple of 7
  if (streak > 0 && streak % 7 === 0) {
    const lastMission = [...quest_history].reverse()
      .find(q => q.quest_type === 'mission')
    if (!lastMission || daysSince(lastMission.completed_at) >= 7) {
      return 'mission'
    }
  }
  // Challenge — level 10+, 20% chance
  if (level >= 10 && Math.random() < 0.20) return 'challenge'
  // Hidden — 15% chance after last Excellent
  const lastQuest = quest_history[quest_history.length - 1]
  if (lastQuest?.grade === 'Excellent' && Math.random() < 0.15) return 'hidden'
  // Prestige — level 100+, 10% chance
  if (level >= 100 && Math.random() < 0.10) return 'prestige'
  return 'daily'
}
export function calculateDifficulty(level) {
  return Math.max(1, Math.ceil(level / 5))
}

export function calculateStreakMultiplier(streak) {
  if (streak >= 100) return 3.0
  if (streak >= 60)  return 2.5
  if (streak >= 30)  return 2.0
  if (streak >= 14)  return 1.75
  if (streak >= 7)   return 1.5
  return 1.0
}
export function calculateFinalEXP(grade, streak, prestigeBonus = 1.0) {
  const base = GRADE_BASE_EXP[grade] || 0
  return Math.round(base * calculateStreakMultiplier(streak) * prestigeBonus)
}
export function calculateExpToNext(level) {
  return Math.round(30 * level * (1 + Math.log(level + 1)))
}
export function checkLevelUp(currentLevel, currentExp, addedExp) {
  let level = currentLevel
  let exp = currentExp + addedExp
  let didLevelUp = false
  while (true) {
    const required = calculateExpToNext(level)
    if (exp >= required) {
      exp -= required
      level++
      didLevelUp = true
    } else break
  }
  return {
    didLevelUp,
    newLevel: level,
    newExp: exp,
    newExpToNext: calculateExpToNext(level),
    newTitle: didLevelUp ? getTitleForLevel(level) : null
  }
}

const OVERCOME_THRESHOLD = 10 // positive signals in last 14 log entries
export function checkWeaknessOvercome(profile) {
  const weaknessStat = WEAKNESS_MAP[profile.weakness_signal]
  if (!weaknessStat) return null
  const last14 = profile.behavior_log.slice(-14)
  const positiveSignals = last14.filter(e => {
    if (weaknessStat === 'focus') return e.response_depth === 'thorough'
    if (weaknessStat === 'consistency') return e.completedSameDay
    if (weaknessStat === 'adaptability') return e.difficulty_handled === 'exceeded'
    if (weaknessStat === 'execution') return e.completion_quality === 'strong'
    return false
  }).length
  if (positiveSignals >= OVERCOME_THRESHOLD) {
    return {
      overcome: true,
      stat: weaknessStat,
      bonus: 3 // permanent +3 to that stat
    }
  }
  return null
}

export function detectUserState(profile) {
  const recentLog    = profile.behavior_log.slice(-10)
  const recentGrades = profile.quest_history.slice(-7).map(q => q.grade)
  const daysSinceActive = daysSince(profile.last_active)
  const scores = {
    bored: 0, burned_out: 0, stagnant: 0,
    struggling: 0, returning: 0, peaking: 0, optimal: 0
  }
  // BORED
  const excellentTrail = getTrailingCount(recentGrades, 'Excellent')
  if (excellentTrail >= 5) scores.bored += 3
  const rushed = recentLog.filter(e => e.response_depth === 'rushed').length
  if (rushed >= 3) scores.bored += 2
  // BURNED_OUT
  const failCount = recentGrades.filter(g => g === 'Fail').length
  if (failCount >= 3) scores.burned_out += 3
  if (daysSinceActive >= 2 && daysSinceActive < 7) scores.burned_out += 2
  // STAGNANT
  const basicCount = recentGrades.filter(g => g === 'Basic').length
  if (basicCount >= 4) scores.stagnant += 2
  // STRUGGLING
  if (failCount >= 4) scores.struggling += 4
  const badCount = recentGrades.filter(g => g === 'Bad').length
  if (badCount + failCount >= 5) scores.struggling += 3
  // RETURNING
  if (daysSinceActive >= 3) scores.returning += 5
  // PEAKING
  if (excellentTrail >= 3 && profile.streak >= 7) scores.peaking += 4
  scores.optimal += 1
  const dominant = Object.entries(scores).sort(([,a],[,b]) => b - a)[0]
  return { state: dominant[0], confidence: dominant[1] }
}

Queue response locally, retry silently on next load

Use generic template: "Something that was always in you has finally surfaced."

Keep cached summary

Skip silently — no question shown today
